define HSV_to_RGB():
  if ((Hue > "360") or (Hue = "360")):
    Hue = (Hue mod 360)
  V = (Value / 255)
  S = (Saturation / 255)
  C = (V * S)
  H' = (Hue / 60)
  X = (C * (1 - abs(((H' mod 2) - 1))))
  m = (V - C)
  if ((H' = "0") or ((H' > "0") and (H' < "1"))):
    R = C
    G = X
    B = "0"
  if ((H' = "1") or ((H' > "1") and (H' < "2"))):
    R = X
    G = C
    B = "0"
  if ((H' = "2") or ((H' > "2") and (H' < "3"))):
    R = "0"
    G = C
    B = X
  if ((H' = "3") or ((H' > "3") and (H' < "4"))):
    R = "0"
    G = X
    B = C
  if ((H' = "4") or ((H' > "4") and (H' < "5"))):
    R = X
    G = "0"
    B = C
  if ((H' = "5") or ((H' > "5") and (H' < "6"))):
    R = C
    G = "0"
    B = X
  R = round(((R + m) * 255))
  G = round(((G + m) * 255))
  B = round(((B + m) * 255))
